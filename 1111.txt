  <script>
    // 預設常見品項 (含在地美食與超商)
    const defaultData = {
      breakfast: ['蛋餅', '豬肉蛋吐司', '飯糰', '燒餅油條', '7-11 飯糰', '全家 匠吐司', '鐵板麵', '蘿蔔糕', '燕麥麥片', '地瓜+無糖豆漿'],
      lunch: ['排骨便當', '雞腿飯', '牛肉麵', '水餃', '7-11 滷肉飯', '全家 涼麵', '乾麵加湯', '健康餐盒', '火鍋', '義大利麵'],
      dinner: ['滷味', '鹽酥雞', '迴轉壽司', '日式拉麵', '燒肉丼飯', '7-11 關東煮', '小火鍋', '鐵板燒', '麻辣燙', '鹹水雞']
    };

    // 色彩盤 (簡約柔和風)
    const colors = [
      '#bfdbfe', '#93c5fd', '#60a5fa', '#fef08a', 
      '#fde047', '#fed7aa', '#fbcfe8', '#a7f3d0'
    ];

    let currentMode = 'breakfast';
    let foodLists = {};
    let isSpinning = false;
    let currentRotation = 0;

    const canvas = document.getElementById('wheelCanvas');
    const ctx = canvas.getContext('2d');

    // 1. 初始化資料 (讀取 LocalStorage 或預設值)
    function initData() {
      const saved = localStorage.getItem('food_wheel_data');
      if (saved) {
        try {
          foodLists = JSON.parse(saved);
        } catch(e) {
          foodLists = JSON.parse(JSON.stringify(defaultData));
        }
      } else {
        foodLists = JSON.parse(JSON.stringify(defaultData));
      }
    }

    function saveData() {
      localStorage.setItem('food_wheel_data', JSON.stringify(foodLists));
    }

    // 2. 切換模式
    function switchMode(mode) {
      if (isSpinning) return;
      currentMode = mode;

      // 更新按鈕樣式
      document.querySelectorAll('.mode-btn').forEach(btn => {
        btn.classList.remove('active');
      });
      event.target.classList.add('active');

      // 更新介面顯示
      const addForm = document.getElementById('addForm');
      const itemList = document.getElementById('itemList');
      const comboInfo = document.getElementById('comboInfo');
      const listTitle = document.getElementById('listTitle');

      if (mode === 'combo') {
        listTitle.innerText = '套餐規則';
        addForm.style.display = 'none';
        itemList.style.display = 'none';
        comboInfo.style.display = 'block';
        document.getElementById('resultTitle').innerText = '準備好了嗎？';
        document.getElementById('resultValue').innerText = '點擊開始抽一日套餐';
      } else {
        const titles = { breakfast: '早餐清單', lunch: '午餐清單', dinner: '晚餐清單' };
        listTitle.innerText = titles[mode];
        addForm.style.display = 'flex';
        itemList.style.display = 'block';
        comboInfo.style.display = 'none';
        document.getElementById('resultTitle').innerText = '按下按鈕開始旋轉';
        document.getElementById('resultValue').innerText = '？';
        renderList();
      }

      drawWheel();
    }

    // 3. 繪製輪盤
    function drawWheel() {
      const items = currentMode === 'combo' ? ['早', '午', '晚'] : foodLists[currentMode];
      const numItems = items.length;
      const radius = canvas.width / 2;

      ctx.clearRect(0, 0, canvas.width, canvas.height);

      if (numItems === 0) {
        ctx.beginPath();
        ctx.arc(radius, radius, radius, 0, Math.PI * 2);
        ctx.fillStyle = '#e2e8f0';
        ctx.fill();
        ctx.fillStyle = '#64748b';
        ctx.font = '16px sans-serif';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText('請先新增品項', radius, radius);
        return;
      }

      const arc = (Math.PI * 2) / numItems;

      for (let i = 0; i < numItems; i++) {
        const angle = i * arc;
        
        // 畫扇形
        ctx.beginPath();
        ctx.arc(radius, radius, radius, angle, angle + arc);
        ctx.lineTo(radius, radius);
        ctx.fillStyle = colors[i % colors.length];
        ctx.fill();
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 2;
        ctx.stroke();

        // 畫文字
        ctx.save();
        ctx.translate(radius, radius);
        ctx.rotate(angle + arc / 2);
        ctx.fillStyle = '#1e293b';
        ctx.font = '500 14px sans-serif';
        ctx.textAlign = 'right';
        ctx.textBaseline = 'middle';
        
        // 文字截斷處理
        let text = items[i];
        if (text.length > 8) text = text.substring(0, 7) + '...';
        ctx.fillText(text, radius - 20, 0);
        ctx.restore();
      }
    }

    // 4. 加密級高隨機度隨機抽取函數
    function getRandomIndex(max) {
      const array = new Uint32Array(1);
      window.crypto.getRandomValues(array);
      return array[0] % max;
    }

    // 5. 開始轉動
    function startSpin() {
      if (isSpinning) return;

      if (currentMode === 'combo') {
        spinCombo();
        return;
      }

      const items = foodLists[currentMode];
      if (!items || items.length === 0) {
        alert('清單中沒有品項，請先新增！');
        return;
      }

      isSpinning = true;
      document.getElementById('spinBtn').disabled = true;
      document.getElementById('resultTitle').innerText = '交給命運決定中...';
      document.getElementById('resultValue').innerText = '🌀';

      const selectedIndex = getRandomIndex(items.length);
      const numItems = items.length;
      const arc = (Math.PI * 2) / numItems;

      // 計算目標角度 (指針在頂部 270度 / 1.5PI 位置)
      const targetAngle = (3 * Math.PI / 2) - (selectedIndex * arc + arc / 2);
      const extraRounds = (Math.floor(Math.random() * 3) + 5) * Math.PI * 2; // 轉 5~7 圈
      const totalRotation = currentRotation + extraRounds + (targetAngle - (currentRotation % (Math.PI * 2)));

      const duration = 4000; // 4 秒
      const start = performance.now();
      const initialRotation = currentRotation;

      function animate(time) {
        let elapsed = time - start;
        let progress = Math.min(elapsed / duration, 1);
        
        // Ease-out 減速效果
        let easeOut = 1 - Math.pow(1 - progress, 3);
        currentRotation = initialRotation + (totalRotation - initialRotation) * easeOut;

        canvas.style.transform = `rotate(${currentRotation}rad)`;

        if (progress < 1) {
          requestAnimationFrame(animate);
        } else {
          isSpinning = false;
          document.getElementById('spinBtn').disabled = false;
          const result = items[selectedIndex];
          document.getElementById('resultTitle').innerText = '今天吃這個！';
          document.getElementById('resultValue').innerText = result;
          
          // 彩帶效果
          confetti({ particleCount: 60, spread: 60, origin: { y: 0.7 } });
        }
      }

      requestAnimationFrame(animate);
    }

    // 6. 套餐模式專用抽獎邏輯 (防止三餐重複)
    function spinCombo() {
      isSpinning = true;
      document.getElementById('spinBtn').disabled = true;
      document.getElementById('resultTitle').innerText = '正在配對完美的精選三餐...';
      document.getElementById('resultValue').innerText = '🎲';

      let bList = foodLists.breakfast;
      let lList = foodLists.lunch;
      let dList = foodLists.dinner;

      if (bList.length === 0 || lList.length === 0 || dList.length === 0) {
        alert('請確保早餐、午餐、晚餐清單中都至少有一個品項！');
        isSpinning = false;
        document.getElementById('spinBtn').disabled = false;
        return;
      }

      // 自動重抽機制：直到三個品項都不相同
      let bRes, lRes, dRes;
      let attempts = 0;
      do {
        bRes = bList[getRandomIndex(bList.length)];
        lRes = lList[getRandomIndex(lList.length)];
        dRes = dList[getRandomIndex(dList.length)];
        attempts++;
      } while ((bRes === lRes || lRes === dRes || bRes === dRes) && attempts < 100);

      // 快速旋轉動畫效果
      let spins = 0;
      const timer = setInterval(() => {
        currentRotation += 0.5;
        canvas.style.transform = `rotate(${currentRotation}rad)`;
        spins++;
        if (spins > 30) {
          clearInterval(timer);
          isSpinning = false;
          document.getElementById('spinBtn').disabled = false;
          document.getElementById('resultTitle').innerText = '✨ 今日精選套餐 ✨';
          document.getElementById('resultValue').innerHTML = `
            <div style="font-size: 1rem; text-align: left; margin-top: 5px; line-height: 1.8;">
              🍳 <b>早餐：</b>${bRes}<br>
              🍱 <b>午餐：</b>${lRes}<br>
              🍜 <b>晚餐：</b>${dRes}
            </div>
          `;
          confetti({ particleCount: 100, spread: 80, origin: { y: 0.6 } });
        }
      }, 50);
    }

    // 7. 管理品項列表 UI
    function renderList() {
      const itemList = document.getElementById('itemList');
      itemList.innerHTML = '';
      const items = foodLists[currentMode] || [];

      items.forEach((item, index) => {
        const li = document.createElement('li');
        li.className = 'item-tag';
        li.innerHTML = `
          <span>${item}</span>
          <button class="delete-btn" onclick="deleteItem(${index})">✕</button>
        `;
        itemList.appendChild(li);
      });
    }

    function addItem() {
      const input = document.getElementById('newItemInput');
      const val = input.value.trim();
      if (!val) return;

      foodLists[currentMode].push(val);
      saveData();
      renderList();
      drawWheel();
      input.value = '';
    }

    function deleteItem(index) {
      foodLists[currentMode].splice(index, 1);
      saveData();
      renderList();
      drawWheel();
    }

    function handleKeyPress(e) {
      if (e.key === 'Enter') addItem();
    }

    function resetCurrentList() {
      if (currentMode === 'combo') return;
      if (confirm(`確定要將${document.getElementById('listTitle').innerText}恢復為預設嗎？`)) {
        foodLists[currentMode] = [...defaultData[currentMode]];
        saveData();
        renderList();
        drawWheel();
      }
    }

    // 8. 頁面載入初始化
    initData();
    renderList();
    drawWheel();
  </script>
